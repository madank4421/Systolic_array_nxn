module systolic_array_controller #(
  parameter ARRAY_SIZE = 4,
  parameter DATA_WIDTH = 8
)(
  input logic [DATA_WIDTH-1:0] matrix_a [0:ARRAY_SIZE-1][0:ARRAY_SIZE-1],
  input logic [DATA_WIDTH-1:0] matrix_b [0:ARRAY_SIZE-1][0:ARRAY_SIZE-1],
  input logic [2:0] active_size,
  input logic start,
  input logic clk,
  input logic rst,
  output logic busy,
  output logic done,
  output logic [DATA_WIDTH-1:0] row_stream [0:ARRAY_SIZE-1],
  output logic [DATA_WIDTH-1:0] col_stream [0:ARRAY_SIZE-1]
);

  typedef enum logic [1:0] {IDLE, LOAD, FLUSH, DONE} state_t;
  state_t curr_state, next_state;

  localparam int COUNTER_WIDTH = $clog2(3 * ARRAY_SIZE);
  logic [COUNTER_WIDTH-1:0] cycle_count;

  always_ff @(posedge clk) begin
    if (rst)
      curr_state <= IDLE;
    else
      curr_state <= next_state;
  end

  always_comb begin
    next_state = curr_state;
    case (curr_state)
      IDLE: begin
        if (start)
          next_state = LOAD;
      end
      LOAD: begin
        if (cycle_count == ((2 * active_size) - 2))
          next_state = FLUSH;
      end
      FLUSH: begin
        if (cycle_count == ((3 * active_size) - 3))
          next_state = DONE;
      end
      DONE: begin
        next_state = IDLE;
      end
      default: begin
        next_state = IDLE;
      end
    endcase
  end

  assign busy = (curr_state != IDLE);
  assign done = (curr_state == DONE);

  always_ff @(posedge clk) begin
    if (rst)
      cycle_count <= '0;
    else if ((curr_state == LOAD) || (curr_state == FLUSH))
      cycle_count <= cycle_count + 1'b1;
    else
      cycle_count <= '0;
  end

  integer i;
  logic signed [COUNTER_WIDTH:0] col;

  always_comb begin
    for (i = 0; i < ARRAY_SIZE; i++) begin
      row_stream[i] = '0;
      col_stream[i] = '0;
    end

    if (curr_state == LOAD) begin
      for (i = 0; i < ARRAY_SIZE; i++) begin
        col = cycle_count - i;
        if (i < active_size) begin
          if ((col >= 0) && (col < active_size)) begin
            row_stream[i] = matrix_a[i][col];
            col_stream[i] = matrix_b[col][i];
          end
        end
      end
    end
  end

endmodule
